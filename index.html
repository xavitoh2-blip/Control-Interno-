function doGet(e) { var accion = e.parameter.accion; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (accion === "verEmpleados") { var sheet = ss.getSheetByName("Nombres"); var datos = sheet ? sheet.getDataRange().getValues() : []; return ContentService.createTextOutput(JSON.stringify(datos)) .setMimeType(ContentService.MimeType.JSON); } if (accion === "verProductosInventario") { var sheet = ss.getSheetByName("ProductosInventario"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } var datos = sheet.getDataRange().getValues(); return ContentService.createTextOutput(JSON.stringify(datos)) .setMimeType(ContentService.MimeType.JSON); } if (accion === "verTareas") { var sheet = ss.getSheetByName("Tareas_Configuracion") || ss.getSheetByName("TareasConfig"); var datos = sheet ? sheet.getDataRange().getValues() : []; return ContentService.createTextOutput(JSON.stringify(datos)) .setMimeType(ContentService.MimeType.JSON); } if (accion === "verAsistencia") { var sheet = ss.getSheetByName("Asistencias"); var datos = sheet ? sheet.getDataRange().getValues() : []; return ContentService.createTextOutput(JSON.stringify(datos)) .setMimeType(ContentService.MimeType.JSON); } if (accion === "verCuadreInventario") { var sheet = ss.getSheetByName("Cuadresinventario") || ss.getSheetByName("Cuadresinven"); var datos = sheet ? sheet.getDataRange().getValues() : []; return ContentService.createTextOutput(JSON.stringify(datos)) .setMimeType(ContentService.MimeType.JSON); } if (accion === "verPedidos") { var sheet = ss.getSheetByName("PEDIDOS"); var datos = sheet ? sheet.getDataRange().getValues() : []; return ContentService.createTextOutput(JSON.stringify(datos)) .setMimeType(ContentService.MimeType.JSON); } // Obtenemos los productos que ya fueron contados hoy para evitar duplicados entre dispositivos if (accion === "verProductosContadosHoy") { var sheet = ss.getSheetByName("Cuadresinventario") || ss.getSheetByName("Cuadresinven"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } var datos = sheet.getDataRange().getValues(); var fechaHoy = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd"); var usuario = e.parameter.empleado; var contados = []; // Recorrer filas omitiendo encabezados (asumiendo Fecha en Col 0, Empleado en Col 3, Producto en Col 4) for (var i = 1; i < datos.length; i++) { var filaFecha = datos[i][0]; var filaEmpleado = datos[i][3]; var filaProducto = datos[i][4]; // Formatear la fecha guardada para comparar if (filaFecha instanceof Date) { filaFecha = Utilities.formatDate(filaFecha, Session.getScriptTimeZone(), "yyyy-MM-dd"); } if (filaFecha === fechaHoy && (!usuario || filaEmpleado === usuario)) { contados.push(filaProducto); } } return ContentService.createTextOutput(JSON.stringify(contados)) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "Accion no reconocida" })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var accion = e.parameter.accion; var ss = SpreadsheetApp.getActiveSpreadsheet(); var fecha = new Date(); var fechaStr = Utilities.formatDate(fecha, Session.getScriptTimeZone(), "yyyy-MM-dd"); var horaStr = Utilities.formatDate(fecha, Session.getScriptTimeZone(), "HH:mm:ss"); if (accion === "Asistencia") { var sheet = ss.getSheetByName("Asistencias") || ss.insertSheet("Asistencias"); if (sheet.getLastRow() === 0) { sheet.appendRow(["Fecha", "Hora", "Sucursal", "Turno", "Tipo", "Empleado", "Horas Extra"]); } sheet.appendRow([ fechaStr, horaStr, e.parameter.sucursal || "", e.parameter.turno || "", e.parameter.tipo || "", e.parameter.empleado || "", e.parameter.horasExtra || 0 ]); return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); } if (accion === "CuadreInventario") { var sheet = ss.getSheetByName("Cuadresinventario") || ss.getSheetByName("Cuadresinven") || ss.insertSheet("Cuadresinventario"); if (sheet.getLastRow() === 0) { sheet.appendRow(["Fecha", "Hora", "Sucursal", "Empleado", "Producto", "Sistema", "Fisico", "Diferencia", "Observacion"]); } sheet.appendRow([ fechaStr, horaStr, e.parameter.sucursal || "", e.parameter.empleado || "", e.parameter.producto || "", e.parameter.sistema || 0, e.parameter.fisico || 0, e.parameter.diferencia || 0, e.parameter.observacion || "" ]); return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); } if (accion === "Tareas") { var sheet = ss.getSheetByName("TAREAS") || ss.insertSheet("TAREAS"); if (sheet.getLastRow() === 0) { sheet.appendRow(["Fecha", "Hora", "Sucursal", "Empleado", "Tarea", "Estado"]); } sheet.appendRow([ fechaStr, horaStr, e.parameter.sucursal || "", e.parameter.empleado || "", e.parameter.tarea || "", e.parameter.estado || "" ]); return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); } if (accion === "Pedidos") { var sheet = ss.getSheetByName("PEDIDOS") || ss.insertSheet("PEDIDOS"); if (sheet.getLastRow() === 0) { sheet.appendRow(["Fecha", "Hora", "Sucursal", "Empleado", "Producto", "Cantidad", "Estado"]); } sheet.appendRow([ fechaStr, horaStr, e.parameter.sucursal || "", e.parameter.empleado || "", e.parameter.producto || "", e.parameter.cantidad || 0, "Pendiente" ]); return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); } return ContentService.createTextOutput("Acción desconocida").setMimeType(ContentService.MimeType.TEXT); }
